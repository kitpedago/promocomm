	Operation.curPourcentageHF AS PourcentageHF,	
	Operation.Libelle AS Programme,	
	StructureJuridique.RS AS SCCV,	
	Tranche.StadePreviCOM as Lancement_Comm,	
	Tranche.StadeCOM as Previ_Lancement_Comm,	
	Tranche.StadeOS AS Date_OS,	
	Tranche.StadePreviOS AS Previ_OS,	
	Tranche.DateLivraisonContractuelle AS Date_livraison,	
	Tranche.NbLot AS Nb_logts,	
	Tranche.NbLgtPSLA AS Nb_logts_PSLA,	
	Tranche.NbLgtHorsPSLA AS Nb_logts_Hors_PSLA,	
	Tranche.NbResa AS Nb_resa,	
	( Tranche.NbLot - Tranche.NbResa )  AS Invendus,	
	Tranche.NbResa_N AS Nb_resa_annee_en_cours,	
	Tranche.NbResa_N_PSLA AS Nb_resa_PSLA_annee_en_cours,	
	Tranche.NbActeVEFA AS Nb_acte_VEFA,	
	Tranche.NbActeVEFA_N AS Nb_acte_VEFA_Annee_En_Cours,	
	( Tranche.NbActeVEFA + Tranche.NbActeVEFA_N )  AS Total_Actes,	
	Tranche.NbLeveeOption AS NbLeveeOption,	
	Tranche.NbPhaseLoc AS NbPhaseLoc,	
	Tranche.NbLeveeOption_N AS NbLeveeOption_annee_en_cours,	
	( ( Tranche.NbLgtHorsPSLA - Tranche.NbResa_N ) + Tranche.NbPhaseLoc )  AS TotalResteASigner,	
	( Tranche.NbLgtHorsPSLA - Tranche.NbResa_N )  AS DontLocAccession,	
	( Tranche.NbResa_N * Operation.curPourcentageHF )  AS ProrataSCCVResa,	
	( Tranche.NbResa_N_PSLA * Operation.curPourcentageHF )  AS ProrataSCCVPSLA,	
	( Tranche.NbActeVEFA_N * Operation.curPourcentageHF )  AS ProrataSCCVActe,	
	( Tranche.NbLeveeOption * Operation.curPourcentageHF )  AS ProrataSCCVLeveeOption
FROM 
	(
		StructureJuridique
		INNER JOIN
		(
			Associe
			RIGHT OUTER JOIN
			Operation
			ON Associe.IDAssocie = Operation.curIDAssocieHorsHF
		)
		ON StructureJuridique.IDStructureJuridique = Operation.IDStructureJuridique
	)
	INNER JOIN
	Tranche
	ON Operation.IDOperation = Tranche.IDOperation
WHERE 
	(
	Operation.MasquerCommercial = 0
)

	IIf(NOT (ISNULL(tStadeAvancement.DateReelle)),'Date réelle',IIf(NOT (ISNULL(tStadeAvancement.DatePrevMAJPromo)),'Date prévisionnelle',NULL)) as TypeDateStadeAvancement

	tListeAvancement,	
	tStadeAvancement,
	tOperation,
	tTranche

	tListeAvancement.IDListeAvancement = tStadeAvancement.IDListeAvancement AND
